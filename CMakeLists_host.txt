# Host (x86) SDL2 simulation build
cmake_minimum_required(VERSION 3.10)
project(my_test_sim C)

set(CMAKE_C_STANDARD 11)

# ========== Host compiler ==========
set(CMAKE_C_COMPILER gcc)

# ========== SDL2 ==========
find_package(SDL2 REQUIRED)
include_directories(${SDL2_INCLUDE_DIRS})

# ========== Compile flags ==========
set(CMAKE_C_FLAGS "${CMAKE_C_FLAGS} -Wall -O0 -g -DHOST_SIMULATION=1")
set(CMAKE_EXE_LINKER_FLAGS "${CMAKE_EXE_LINKER_FLAGS} -lpthread -lm ${SDL2_LIBRARIES}")

# ========== Include dirs ==========
include_directories(
    ${CMAKE_CURRENT_SOURCE_DIR}
    ${CMAKE_CURRENT_SOURCE_DIR}/hal
    ${CMAKE_CURRENT_SOURCE_DIR}/services
    ${CMAKE_CURRENT_SOURCE_DIR}/ui
    ${CMAKE_CURRENT_SOURCE_DIR}/web
    ${CMAKE_CURRENT_SOURCE_DIR}/storage
    ${CMAKE_CURRENT_SOURCE_DIR}/infra
    ${CMAKE_CURRENT_SOURCE_DIR}/lvgl_src
    ${CMAKE_CURRENT_SOURCE_DIR}/lvgl_src/src
)

# ========== LVGL host config ==========
configure_file(${CMAKE_CURRENT_SOURCE_DIR}/lv_conf_host.h
               ${CMAKE_CURRENT_SOURCE_DIR}/lvgl_src/lv_conf.h COPYONLY)

# ========== LVGL library ==========
file(GLOB_RECURSE LVGL_SOURCES ${CMAKE_CURRENT_SOURCE_DIR}/lvgl_src/src/*.c)
list(FILTER LVGL_SOURCES EXCLUDE REGEX ".*/test/.*")
list(FILTER LVGL_SOURCES EXCLUDE REGEX ".*/wayland/.*")
list(FILTER LVGL_SOURCES EXCLUDE REGEX ".*/x11/.*")
list(FILTER LVGL_SOURCES EXCLUDE REGEX ".*/windows/.*")
list(FILTER LVGL_SOURCES EXCLUDE REGEX ".*/nuttx/.*")

add_library(lvgl_host ${LVGL_SOURCES})
target_include_directories(lvgl_host PUBLIC
    ${CMAKE_CURRENT_SOURCE_DIR}/lvgl_src
    ${CMAKE_CURRENT_SOURCE_DIR}/lvgl_src/src
    ${SDL2_INCLUDE_DIRS}
)
target_compile_definitions(lvgl_host PRIVATE LV_CONF_INCLUDE_SIMPLE=1)

# ========== Application: stub-only sources for host ==========
set(APP_SOURCES
    main.c
    sqlite3.c

    # SDL display backend (replaces DRM)
    hal/display_sdl.c

    # All HAL/Service stubs
    hal/rs485_uart_stub.c
    hal/can_socket_stub.c
    services/mqtt_client_stub.c

    # Core services (no hardware dependency)
    services/data_recorder.c
    services/data_bus.c
    services/modbus_master.c
    services/can_manager.c

    # UI
    ui/ui_dashboard.c
    ui/ui_ota.c
    ui/ui_page_mqtt.c
    ui/ui_page_modbus.c
    ui/ui_page_can.c

    # OTA (uses HTTP, needs network stub too)
    ota_manager.c

    # Infra
    infra/logger.c

    # Stubs for hardware/network services
    services/mqtt_client_stub.c
)

# Create stubs inline for services that need networking
# ntp_sync stub
file(WRITE ${CMAKE_CURRENT_BINARY_DIR}/ntp_sync_stub.c
     "#include \"ntp_sync.h\"\n#include <stdio.h>\nint ntp_sync_init(void){printf(\"NTP stub: skipped\\n\");return 0;}\nvoid ntp_sync_stop(void){}\n")
# web_server stub
file(WRITE ${CMAKE_CURRENT_BINARY_DIR}/web_server_stub.c
     "#include \"web_server.h\"\n#include <stdio.h>\nint web_server_start(int port){(void)port;printf(\"HTTP stub: port %d (not started)\\n\",port);return 0;}\nvoid web_server_stop(void){}\nvoid web_server_update_data(float t,float h,bool v){(void)t;(void)h;(void)v;}\n#include <stdbool.h>\n")
# watchdog stub
file(WRITE ${CMAKE_CURRENT_BINARY_DIR}/watchdog_stub.c
     "#include \"watchdog.h\"\n#include <stdio.h>\nint watchdog_init(int t){(void)t;printf(\"Watchdog stub: disabled\\n\");return 0;}\nvoid watchdog_stop(void){}\n")
# database stub (SQLite amalgamation might be too big; use simple stub)
file(WRITE ${CMAKE_CURRENT_BINARY_DIR}/database_stub.c
     "#include \"database.h\"\n#include <stdio.h>\nint database_init(void){printf(\"DB stub: memory only\\n\");return 0;}\nvoid database_close(void){}\nvoid database_insert_sensor_data(float t,float h,int v){(void)t;(void)h;(void)v;}\nint database_get_sensor_data(int h,sensor_data_t *d,int m){if(d)d[0].timestamp=0;return 0;}\n")
# ota stub
file(WRITE ${CMAKE_CURRENT_BINARY_DIR}/ota_manager_stub.c
     "#include \"ota_manager.h\"\n#include <stdio.h>\nvoid ota_init(const char*s){printf(\"OTA stub: %s\\n\",s);}\nvoid ota_set_type(ota_type_t t){(void)t;}\nvoid ota_set_app_install_path(const char*p){(void)p;}\n")
# config stub
file(WRITE ${CMAKE_CURRENT_BINARY_DIR}/config_file_stub.c
     "#include \"config_file.h\"\n#include <string.h>\nint config_load(const char *path, config_t *cfg) { memset(cfg,0,sizeof(*cfg)); return 0; }\nint config_save(const char *path, const config_t *cfg) { (void)path; (void)cfg; return 0; }\n")
# api stubs
file(WRITE ${CMAKE_CURRENT_BINARY_DIR}/api_system_stub.c
     "#include \"api_system.h\"\n")
file(WRITE ${CMAKE_CURRENT_BINARY_DIR}/api_ota_stub.c
     "#include \"api_ota_web.h\"\n")
file(WRITE ${CMAKE_CURRENT_BINARY_DIR}/api_device_stub.c
     "#include \"api_device.h\"\n")

list(APPEND APP_SOURCES
    ${CMAKE_CURRENT_BINARY_DIR}/ntp_sync_stub.c
    ${CMAKE_CURRENT_BINARY_DIR}/web_server_stub.c
    ${CMAKE_CURRENT_BINARY_DIR}/watchdog_stub.c
    ${CMAKE_CURRENT_BINARY_DIR}/database_stub.c
    ${CMAKE_CURRENT_BINARY_DIR}/ota_manager_stub.c
    ${CMAKE_CURRENT_BINARY_DIR}/config_file_stub.c
    ${CMAKE_CURRENT_BINARY_DIR}/api_system_stub.c
    ${CMAKE_CURRENT_BINARY_DIR}/api_ota_stub.c
    ${CMAKE_CURRENT_BINARY_DIR}/api_device_stub.c
)

add_executable(my_test_sim ${APP_SOURCES})
target_link_libraries(my_test_sim lvgl_host ${SDL2_LIBRARIES} pthread m)
